REFLsep =
fun (M Lev g : Syn.form) (n : nat) =>
Prov.prov Elim.ZFpure
  (Rel2.chain (Rel2.G M (S n))
     (Syn.Xf
        (Syn.Kf (Syn.Kf Lev (Syn.Mem 1 0))
           (Syn.All
              (Syn.Imp (Syn.Mem 0 1)
                 (Syn.Qf (Rel2.rel M (Syn.ren sigS g))
                    (relU 1 (Syn.ren sigS g))))))))
     : Syn.form -> Syn.form -> Syn.form -> nat -> Prop

DEFsep =
fun (M Lev g : Syn.form) (n : nat) =>
Prov.prov Elim.ZFpure
  (Rel2.chain (Rel2.G M (S n))
     (Syn.All
        (Syn.Imp Lev
           (Syn.Imp (Syn.Mem 1 0)
              (Syn.Xf
                 (Syn.Kf M
                    (Syn.All
                       (Syn.Qf (Syn.Mem 0 1)
                          (Syn.Kf (Syn.Mem 0 3) (relU 2 (Syn.ren sigD g)))))))))))

STR =
fun Lev : Syn.form =>
Syn.All
  (Syn.Imp Lev
     (Syn.All
        (Syn.Imp (Syn.Mem 0 1)
           (Syn.All (Syn.Imp (Syn.Mem 0 1) (Syn.Mem 0 2))))))
     : Syn.form -> Syn.form

Arguments STR Lev
SSUB =
fun M Lev : Syn.form =>
Syn.All (Syn.Imp Lev (Syn.All (Syn.Imp (Syn.Mem 0 1) M)))
     : Syn.form -> Syn.form -> Syn.form

Arguments SSUB M Lev
SIN =
fun M Lev : Syn.form => Syn.All (Syn.Imp Lev M)
     : Syn.form -> Syn.form -> Syn.form

LSEP =
fun M : Syn.form =>
forall g : Syn.form,
Syn.syms g = nil ->
forall n : nat,
1 <= n ->
Ax.closedk n (Checks.SepF g) = true ->
Prov.prov Elim.ZFpure (Rel2.chain (Rel2.G M n) (Rel2.rel M (Checks.SepF g)))
     : Syn.form -> Prop

Arguments LSEP M
