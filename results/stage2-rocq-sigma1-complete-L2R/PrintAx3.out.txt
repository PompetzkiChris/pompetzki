RFs =
fun M : Syn.form =>
fix RFs (L : list (nat * Syn.form)) : Syn.form :=
  match L with
  | nil => Pure.Top
  | (c :: r)%list => Syn.Kf (RF1 M c) (RFs r)
  end
     : Syn.form -> list (nat * Syn.form) -> Syn.form

RF1 =
fun (M : Syn.form) (c : nat * Syn.form) =>
let (m, phi) := c in
wrapQ 0 m
  (Syn.Qf (Rel2.rel M (Syn.ren (shm m) phi)) (relU m (Syn.ren (shm m) phi)))
     : Syn.form -> nat * Syn.form -> Syn.form

wrapQ =
fix wrapQ (w k : nat) (b : Syn.form) {struct k} : Syn.form :=
  match k with
  | 0 => b
  | S k' => Syn.All (Syn.Imp (Syn.Mem 0 (S w)) (wrapQ (S w) k' b))
  end
     : nat -> nat -> Syn.form -> Syn.form

Arguments wrapQ (w k)%_nat_scope b
shm =
fun m t : nat => if PeanoNat.Nat.ltb t m then t else S (S t)
     : nat -> nat -> nat

SCOV =

Sub =
fun a b : nat => Syn.All (Syn.Imp (Syn.Mem 0 (S a)) (Syn.Mem 0 (S b)))
     : nat -> nat -> Syn.form

relU =
fix relU (w : nat) (f : Syn.form) {struct f} : Syn.form :=
  match f with
  | Syn.Mem x y => Syn.Mem x y
  | Syn.Eqv x y => Syn.Eqv x y
  | Syn.Pr p a => Syn.Pr p a
  | Syn.Neg a => Syn.Neg (relU w a)
  | Syn.Imp a b => Syn.Imp (relU w a) (relU w b)
  | Syn.All a => Syn.All (Syn.Imp (Syn.Mem 0 (S w)) (relU (S w) a))
  end
     : nat -> Syn.form -> Syn.form
